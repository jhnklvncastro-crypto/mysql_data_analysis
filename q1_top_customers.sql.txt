SELECT
    c.customer_unique_id,
    SUM(p.payment_value) AS total_amount_spent
FROM customers AS c
JOIN orders AS o
    ON c.customer_id = o.customer_id
JOIN order_payments AS p
    ON o.order_id = p.order_id
WHERE o.order_status = 'delivered'
GROUP BY
    c.customer_unique_id
ORDER BY
    total_amount_spent DESC
LIMIT 10;
